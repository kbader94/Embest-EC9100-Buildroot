# --- Device / partitions ---
setenv mmcdev 0
setenv mmcpart 1
setenv bootpart ${mmcdev}:${mmcpart}

# --- Addresses ---
setenv kernel_addr_r 0x80800000
setenv fdt_addr_r    0x83000000

# --- DTB ---
setenv fdtfile imx6ul-embest-ec9100.dtb

# --- Set Rootfs + Console (UART3 = ttymxc2) ---
setenv bootargs console=ttymxc2,115200 root=/dev/mmcblk0p2 rootwait rw

echo "Loading kernel and FDT from mmc ${bootpart}..."
if fatload mmc ${bootpart} ${kernel_addr_r} zImage; then
  if fatload mmc ${bootpart} ${fdt_addr_r} ${fdtfile}; then
    echo "Booting zImage with ${fdtfile}..."
    bootz ${kernel_addr_r} - ${fdt_addr_r}
  else
    echo "ERROR: Failed to load FDT ${fdtfile}"
  fi
else
  echo "ERROR: Failed to load zImage"
fi
